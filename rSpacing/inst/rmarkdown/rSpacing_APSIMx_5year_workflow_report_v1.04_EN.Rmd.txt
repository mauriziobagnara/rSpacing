# Run execution and fixture construction

The document contains the complete five-year script. With `params$build_fixture = true`, it runs `Spacing.Tool2`, normalizes the outputs, and saves the canonical fixture. When the parameter is set to `false`, it reuses the existing fixture without rerunning APSIMx.

### What the canonical-fixture builder does

This is the operational core of the document. When `params$build_fixture = true`, it defines the complete 2020-2024 Wheat-Maize-Soybean scenario, writes the request and configuration JSON files, runs `Spacing.Tool2`, imports APSIMx outputs from JSON, CSV and SQLite, builds summary tables and plots, and saves a canonical RDS fixture. The scenario uses mock weather, the synthetic six-layer soil profile, and no crop parameter overrides.

The code is hidden because it is long and primarily implements the workflow. For presentation purposes, the important points are: the configuration is created in R, each run is isolated in its own directory, execution errors are trapped before interpretation, and the normalized fixture preserves both requested management and evidence returned by APSIMx.

safe_numeric_summary <- function(data, table_name) {
  if (is.null(data) || !is.data.frame(data) || nrow(data) == 0L) return(NULL)
  numeric_names <- names(data)[vapply(data, is.numeric, logical(1))]
  if (length(numeric_names) == 0L) return(NULL)
  rows <- lapply(numeric_names, function(column_name) {
    values <- data[[column_name]]
    finite <- values[is.finite(values)]
    data.frame(
      table = table_name,
      variable = column_name,
      n = length(values),
      n_finite = length(finite),
      mean = if (length(finite)) mean(finite) else NA_real_,
      sd = if (length(finite) > 1L) stats::sd(finite) else NA_real_,
      min = if (length(finite)) min(finite) else NA_real_,
      median = if (length(finite)) stats::median(finite) else NA_real_,
      max = if (length(finite)) max(finite) else NA_real_,
      stringsAsFactors = FALSE
    )
  })
  do.call(rbind, rows)
}

safe_date <- function(x) {
  if (inherits(x, "Date")) return(x)
  if (inherits(x, c("POSIXct", "POSIXlt"))) return(as.Date(x))
  x <- as.character(x)
  out <- rep(as.Date(NA), length(x))
  formats <- c("%Y-%m-%d", "%Y-%m-%d %H:%M:%S", "%Y-%m-%dT%H:%M:%S",
               "%d/%m/%Y", "%m/%d/%Y", "%d-%m-%Y", "%Y/%m/%d")
  for (fmt in formats) {
    miss <- is.na(out) & !is.na(x)
    if (!any(miss)) break
    out[miss] <- suppressWarnings(as.Date(x[miss], format = fmt))
  }
  miss <- is.na(out) & grepl("^[0-9]{4}-[0-9]{2}-[0-9]{2}", x)
  if (any(miss)) out[miss] <- suppressWarnings(as.Date(substr(x[miss], 1L, 10L), format = "%Y-%m-%d"))
  out
}

first_column <- function(data, exact = character(), regex = character(), numeric = FALSE) {
  if (!is.data.frame(data) || !ncol(data)) return(NA_character_)
  hits <- exact[exact %in% names(data)]
  if (!length(hits) && length(regex)) {
    hits <- unique(unlist(lapply(regex, function(z) grep(z, names(data), value = TRUE, ignore.case = TRUE))))
  }
  if (numeric && length(hits)) hits <- hits[vapply(data[hits], is.numeric, logical(1))]
  if (length(hits)) hits[[1L]] else NA_character_
}

map_crop <- function(year, crop = NULL) {
  ans <- if (is.null(crop)) rep(NA_character_, length(year)) else as.character(crop)
  if (!length(ans)) ans <- rep(NA_character_, length(year))
  if (length(ans) != length(year)) ans <- rep_len(ans, length(year))
  miss <- is.na(ans) | !nzchar(ans)
  idx <- match(as.integer(year[miss]), rotation_yearly$year)
  ans[miss] <- rotation_yearly$crop[idx]
  ans
}

map_cultivar <- function(year, crop) {
  idx <- match(paste(year, crop), paste(rotation_yearly$year, rotation_yearly$crop))
  rotation_yearly$cultivar[idx]
}

collect_events <- function(event, value_regex, date_regex = c("date", "today"), unit = NA_character_) {
  out <- list(); k <- 0L
  for (table_name in names(all_tables)) {
    d <- all_tables[[table_name]]
    if (!nrow(d)) next
    value_cols <- grep(value_regex, names(d), value = TRUE, ignore.case = TRUE)
    if (!length(value_cols)) next
    date_col <- first_column(d, c("Date", "date", "Clock.Today", "Today"), date_regex)
    year_col <- first_column(d, c("Year", "year"), c("^year$"))
    crop_col <- first_column(d, c("Crop", "crop", "Speciess", "species"), c("crop", "species"))
    dates <- if (!is.na(date_col)) safe_date(d[[date_col]]) else rep(as.Date(NA), nrow(d))
    years <- if (!is.na(year_col)) suppressWarnings(as.integer(d[[year_col]])) else suppressWarnings(as.integer(format(dates, "%Y")))
    crops_here <- map_crop(years, if (!is.na(crop_col)) d[[crop_col]] else NULL)
    for (value_col in value_cols) {
      raw <- d[[value_col]]
      amount <- suppressWarnings(as.numeric(raw))
      if (all(is.na(amount))) {
        keep <- !is.na(raw) & nzchar(as.character(raw))
        amount <- rep(NA_real_, length(raw))
      } else keep <- is.finite(amount) & amount != 0
      if (!any(keep)) next
      k <- k + 1L
      yr <- years[keep]; cr <- crops_here[keep]
      out[[k]] <- data.frame(
        cycle = paste0(yr, "_", cr), year = yr, crop = cr,
        cultivar = map_cultivar(yr, cr), event = event, date = dates[keep],
        amount = amount[keep], unit = unit, evidence_table = table_name,
        evidence_variable = value_col, stringsAsFactors = FALSE
      )
    }
  }
  if (length(out)) unique(do.call(rbind, out)) else empty_events()
}

# The source plans use `Speciess` and `Cultivar`; normalize their keys before joining.
normalise_plan_keys <- function(plan) {
  if (!is.data.frame(plan)) return(data.frame(crop=character(), cultivar=character()))
  out <- plan
  crop_name <- intersect(c("crop", "Crop", "species", "Speciess"), names(out))
  cultivar_name <- intersect(c("cultivar", "Cultivar"), names(out))
  if (!length(crop_name)) out$crop <- NA_character_ else names(out)[match(crop_name[[1L]], names(out))] <- "crop"
  if (!length(cultivar_name)) out$cultivar <- NA_character_ else names(out)[match(cultivar_name[[1L]], names(out))] <- "cultivar"
  out
}

collect_annual <- function(value_regex, statistic = c("max", "last"), exclude_regex = NULL) {
  statistic <- match.arg(statistic)
  out <- list(); k <- 0L
  for (table_name in names(all_tables)) {
    d <- all_tables[[table_name]]
    if (!nrow(d)) next
    value_cols <- grep(value_regex, names(d), value = TRUE, ignore.case = TRUE)
    value_cols <- value_cols[vapply(d[value_cols], is.numeric, logical(1))]
    if (!is.null(exclude_regex)) value_cols <- value_cols[!grepl(exclude_regex, value_cols, ignore.case = TRUE)]
    if (!length(value_cols)) next
    date_col <- first_column(d, c("Date", "date", "Clock.Today", "Today"), c("date", "today"))
    year_col <- first_column(d, c("Year", "year"), c("^year$"))
    crop_col <- first_column(d, c("Crop", "crop", "Speciess", "species"), c("crop", "species"))
    dates <- if (!is.na(date_col)) safe_date(d[[date_col]]) else rep(as.Date(NA), nrow(d))
    years <- if (!is.na(year_col)) suppressWarnings(as.integer(d[[year_col]])) else suppressWarnings(as.integer(format(dates, "%Y")))
    crops_here <- map_crop(years, if (!is.na(crop_col)) d[[crop_col]] else NULL)
    for (value_col in value_cols) {
      tmp <- data.frame(year=years, crop=crops_here, date=dates, value=suppressWarnings(as.numeric(d[[value_col]])))
      tmp <- tmp[is.finite(tmp$value) & !is.na(tmp$year) & tmp$year >= START_YEAR & tmp$year <= END_YEAR, , drop=FALSE]
      if (!nrow(tmp)) next
      groups <- split(tmp, interaction(tmp$year, tmp$crop, drop=TRUE))
      for (z in groups) {
        z <- z[order(z$date, na.last=TRUE), , drop=FALSE]
        val <- if (statistic == "max") max(z$value, na.rm=TRUE) else tail(z$value, 1L)
        k <- k + 1L
        out[[k]] <- data.frame(cycle=paste0(z$year[1], "_", z$crop[1]), year=z$year[1], crop=z$crop[1],
          cultivar=map_cultivar(z$year[1], z$crop[1]), variable=value_col, value=val,
          evidence_table=table_name, stringsAsFactors=FALSE)
      }
    }
  }
  if (length(out)) unique(do.call(rbind, out)) else data.frame(
    cycle=character(), year=integer(), crop=character(), cultivar=character(), variable=character(),
    value=double(), evidence_table=character(), stringsAsFactors=FALSE)
}

annual_yield <- collect_annual("yield|grain.*wt|grain.*weight|harvest.*wt|harvest.*weight", "max",
  "biomass|stover|residue|leaf|stem|root")
annual_soil_c <- collect_annual("soil.*carbon|carbon.*soil|total.*c|organic.*c|soilc|soc", "last",
  "co2|flux|respir|emission|rate")
if (nrow(annual_soil_c)) {
  annual_soil_c <- do.call(rbind, lapply(split(annual_soil_c, interaction(annual_soil_c$evidence_table, annual_soil_c$variable, drop=TRUE)), function(z) {
    z <- z[order(z$year), , drop=FALSE]
    z$baseline_value <- z$value[1L]
    z$change_from_baseline <- z$value - z$baseline_value
    z
  }))
  rownames(annual_soil_c) <- NULL
} else {
  annual_soil_c$baseline_value <- double()
  annual_soil_c$change_from_baseline <- double()
}

layered_soil_c <- data.frame(
  cycle=character(), year=integer(), crop=character(), layer=character(), variable=character(),
  value=double(), evidence_table=character(), stringsAsFactors=FALSE)
for (table_name in names(all_tables)) {
  d <- all_tables[[table_name]]
  if (!nrow(d)) next
  value_cols <- grep("soil.*carbon|carbon.*soil|organic.*c|soilc|soc", names(d), value=TRUE, ignore.case=TRUE)
  value_cols <- value_cols[vapply(d[value_cols], is.numeric, logical(1))]
  layer_col <- first_column(d, c("Layer", "layer", "Depth", "depth", "Thickness", "thickness"), c("layer", "depth", "thickness"))
  if (!length(value_cols) || is.na(layer_col)) next
  date_col <- first_column(d, c("Date", "date", "Clock.Today", "Today"), c("date", "today"))
  year_col <- first_column(d, c("Year", "year"), c("^year$"))

# Initial soil-water profile

### How to read the initial water profile

For each layer, `LL15` is the crop lower limit, `DUL` the drained upper limit, `SAT` saturation and `SW` the initial water content. The physically expected ordering is `LL15 < DUL < SAT`; `SW` indicates where the simulation starts within that storage range.

The table describes initial conditions, not irrigation outcomes. The difference `DUL - LL15` represents plant-available water capacity on a volumetric basis and must be combined with layer thickness to obtain a water depth.

```{r initial-water-profile-table}
water_profile <- x$soil_profile[, c("Layer", "DepthTop", "DepthBottom", "Depth_mid_cm", "LL15", "DUL", "SAT", "SW")]
show_table(water_profile, 100)
```

### How to read the soil-water profile

The vertical axis is reversed so the soil surface is at the top. Curves farther to the right represent greater volumetric water content. The position of `SW` relative to `LL15` and `DUL` shows whether each layer starts relatively dry or wet.

The figure is a starting-condition profile, not a time series. The distance between limits is volumetric and should not be interpreted directly as millimetres without multiplying by layer thickness.

```{r initial-water-profile}
long <- reshape(water_profile[,c("Depth_mid_cm","LL15","DUL","SAT","SW")],
  varying=c("LL15","DUL","SAT","SW"), v.names="value", timevar="variable",
  times=c("LL15","DUL","SAT","SW"), direction="long")
print(ggplot(long,aes(value,Depth_mid_cm,colour=variable)) + geom_path() + geom_point() + scale_y_reverse() +
  labs(title="Initial soil-water profile",x="Volumetric water content",y="Depth (cm)",colour=NULL) + theme_minimal())
```

# Preview of APSIMx tables

## Daily

### How to use the daily-table preview

This is a limited preview of the daily APSIMx report, controlled by `params$show_rows`. It is included for traceability and to expose names, units encoded in names, dates and representative values used by later extraction functions.

The visible rows are not the full simulation and should not be used to calculate totals manually. Cumulative fluxes and final-day stocks must be derived from the complete table, not from this preview.

```{r preview-daily}
show_table(x$tables$daily, params$show_rows)
```

## Crop summary

### How to use the crop-summary preview

This table shows crop-level summaries when exported. It can contain aggregated yield, biomass, phenology or management information, depending on the APSIMx report configuration. The fallback note is intentional and directs the reader to annual-yield or harvest evidence if the table is absent.

Interpret column names and units literally. Do not assume that similarly named variables across tables are identical until their APSIMx definitions are checked.

```{r preview-crop-summary}
if (is.data.frame(x$tables$crop_summary) && nrow(x$tables$crop_summary)) {
  show_table(x$tables$crop_summary, params$show_rows)
} else {
  data.frame(note="crop_summary was not exported or contains no rows; see annual_yield and harvest for crop results")
}
```

## Harvest

### How to use the harvest preview

When present, the harvest table supplies direct evidence of harvest timing and crop outcomes. It is particularly useful for distinguishing a successful crop harvest from a run that produced biomass but did not reach harvest.

If it is absent, the report avoids reconstructing actual harvest dates from management requests. Other tables may still contain yield evidence, but statements about harvest timing remain unsupported.

```{r preview-harvest}
if (is.data.frame(x$tables$harvest) && nrow(x$tables$harvest)) {
  show_table(x$tables$harvest, params$show_rows)
} else {
  data.frame(note="harvest was not exported or contains no rows; actual dates cannot be reported without APSIMx evidence")
}
```

## Control

### How to use the control-table preview

The control report provides low-frequency checkpoints and crop-specific summary columns. It is often preferable to the daily report for end-of-cycle values because it avoids selecting a residual value after crop termination.

Check the checkpoint meaning and the crop-specific column name before quoting a result. Values for Wheat, Maize and Soybean must be taken from their corresponding columns rather than selecting the first matching variable.

```{r preview-control}
show_table(x$tables$control, params$show_rows)
```

## Timeseries

### How to use the timeseries preview

This table exposes any additional time-series export returned by `Spacing.Tool2`. It helps identify variables that are not present in the standard daily or control reports. The preview is diagnostic and does not imply that every column is used elsewhere in the report.

As with the daily preview, only the full table can support cumulative or final-value calculations. Empty output means that no separate timeseries file was exported for this run.

```{r preview-timeseries}
show_table(x$tables$timeseries, params$show_rows)
```

# Variable inventory

## Available variables

### How to read the variable inventory

Each row lists a table, variable name, R data type and number of non-missing values. This inventory is the audit trail used to determine what the current APSIMx configuration actually exported. It is also the safest starting point when extending the workflow to new outputs.

A variable name in the inventory proves availability, not scientific suitability. Its unit, temporal meaning and aggregation rule still need to be established before analysis.

```{r variable-inventory}
show_table(x$audit$variable_inventory, 500)
```

## Coverage and missing variables

### How to read coverage and missing-variable checks

The table groups expected categories such as dates, crops, management events, yield and Soil C and reports whether matching variables were found. The `tables` and `variables` fields identify the evidence. A second table lists only missing categories.

Coverage is based on variable-name patterns, so it is a screening check rather than semantic proof. An available category should still be verified against the underlying APSIMx definition; a missing category means the report cannot support that claim without changing the export configuration.

```{r variable-coverage}
show_table(x$audit$variable_coverage, 100)
missing <- x$audit$variable_coverage[!x$audit$variable_coverage$available,,drop=FALSE]
if (nrow(missing)) show_table(missing,100) else data.frame(note="All expected categories are available")
```

# Run diagnostics

### How to read run diagnostics

This block prints the standard output and standard error captured from `Spacing.Tool2`. An empty `STDERR` together with exit status zero and `success = TRUE` is the expected clean result. Warnings or stack traces should be investigated before scientific interpretation.

Diagnostics are operational evidence. They explain execution failures and data-export problems but are not model outputs and should normally be moved to an appendix during a presentation.

```{r diagnostics}
cat("STDOUT\n",x$diagnostics$stdout,"\n\nSTDERR\n",x$diagnostics$stderr)
```
